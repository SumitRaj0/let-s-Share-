import { Engine, RenderPipeline } from '@snippet/core';

const config = {
    resolution: '4k',
    fpsTarget: 60,
    antiAliasing: true
};

function startApp() {
    const engine = new Engine(config);
    
    engine.on('ready', () => {
        console.log('System online. Ready for input.');
        RenderPipeline.start();
    });
}
